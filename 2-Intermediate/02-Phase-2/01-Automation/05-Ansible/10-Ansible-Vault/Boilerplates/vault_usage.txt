# 🔐 Ansible Vault Boilerplate

# To create an encrypted file:
# ansible-vault create secrets.yml

# To edit an existing encrypted file:
# ansible-vault edit secrets.yml

# Example content of a secrets.yml (once decrypted):
db_password: "SuperSecretPassword123"
api_token: "abc-123-def-456"
ssh_private_key: |
  -----BEGIN RSA PRIVATE KEY-----
  ...
  -----END RSA PRIVATE KEY-----

# -----------------------------------------------------------------------------
# HOW TO USE IN A PLAYBOOK
# -----------------------------------------------------------------------------
# File: site.yml
# 
# - name: Deploy with secrets
#   hosts: all
#   vars_files:
#     - secrets.yml
#   tasks:
#     - name: Print (MASKED) password
#       debug:
#         msg: "The password is {{ db_password }}"
#       no_log: true  # CRITICAL: Prevents secret from appearing in logs
