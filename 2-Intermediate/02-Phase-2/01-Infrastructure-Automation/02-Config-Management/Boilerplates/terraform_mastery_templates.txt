# Terraform Boilerplate: Production Remote Backend

# --- 1. THE BACKEND CONFIGURATION ---
terraform {
  backend "s3" {
    bucket         = "my-enterprise-tfstate-bucket"
    key            = "dev/infrastructure.tfstate"
    region         = "us-east-1"
    encrypt        = true
    dynamodb_table = "terraform-lock-table" # Enables STATE LOCKING
  }

  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }
}

---

# Terraform Boilerplate: Modular Resource Call

# --- 2. THE MODULE INVOCATION ---
module "production_vpc" {
  source = "./modules/vpc-standard" # Path to your module

  # Inputs defined in the module's variables.tf
  vpc_cidr         = "10.10.0.0/16"
  environment      = "production"
  enable_flow_logs = true
}

# --- 3. USING MODULE OUTPUTS ---
resource "aws_instance" "web" {
  ami           = "ami-0c55b159cbfafe1f0"
  instance_type = "t3.medium"
  
  # Link to the module output!
  subnet_id     = module.production_vpc.public_subnets[0]
}

---

# CLI Productivity Table

| Command | Purpose | Essential Flag |
| :--- | :--- | :--- |
| `terraform init` | Download providers and setup backend | `-reconfigure` (Force setup) |
| `terraform plan` | Preview changes without applying | `-out=plan.binary` (Safe deploy) |
| `terraform apply` | Build or update infrastructure | `-auto-approve` (CI/CD only) |
| `terraform fmt` | Automatically fix indentation/style | `-recursive` |
| `terraform state rm` | Stop tracking a resource (without deleting it) | |
