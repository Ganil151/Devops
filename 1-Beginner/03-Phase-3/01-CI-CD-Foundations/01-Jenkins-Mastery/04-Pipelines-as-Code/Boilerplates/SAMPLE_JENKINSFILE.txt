pipeline {
    /* 
       THE GOLDEN JENKINSFILE 
       A robust, declarative template for professional DevOps workflows.
    */
    
    agent any // Defines which node to run on. Use { label 'docker' } for specific executors.

    environment {
        // Define environment-wide variables
        APP_NAME = 'devops-orchestrator'
        ENV = 'Staging'
        DOCKER_IMAGE = "ganil151/${APP_NAME}:latest"
    }

    options {
        // Global configuration for the pipeline
        timeout(time: 1, unit: 'HOURS') 
        timestamps()
        buildDiscarder(logRotator(numToKeepStr: '10'))
    }

    stages {
        stage('🚀 Initialize') {
            steps {
                echo "Starting build for ${APP_NAME} in ${ENV} environment..."
            }
        }

        stage('📦 Source Checkout') {
            steps {
                // In a real repo, this is usually handled by the SCM plugin automatically
                echo "Pulling latest code from Git..."
            }
        }

        stage('🛠️ Build') {
            steps {
                echo "Compiling application..."
                // Example: sh 'mvn clean install' or 'npm run build'
            }
        }

        stage('🧪 Testing') {
            parallel {
                stage('Unit Tests') {
                    steps {
                        echo "Running Unit Tests..."
                        // sh 'pytest' or 'npm test'
                    }
                }
                stage('Security Scan') {
                    steps {
                        echo "Running Trivy Vulnerability Scan..."
                        // sh 'trivy image ${DOCKER_IMAGE}'
                    }
                }
            }
        }

        stage('🚢 Deployment') {
            when {
                branch 'main' // Only deploy if we are on the main branch
            }
            steps {
                echo "Deploying to ${ENV} Kubernetes Cluster..."
                // sh 'kubectl apply -f k8s/deployment.yaml'
            }
        }
    }

    post {
        always {
            echo 'Clearing build workspace...'
            cleanWs()
        }
        success {
            echo 'Deployment successful. Sending notification to Slack...'
        }
        failure {
            echo 'Pipeline failed! Alerting the Engineering team.'
        }
    }
}
