# Production Scaling Boilerplate: Load-Balanced Stack

/* 
   This stack demonstrates how to scale a web service horizontally 
   using a dedicated Nginx Load Balancer.
*/

version: '3.8'

services:
  # Internal Web Service (Hidden from public)
  web:
    image: nginx:alpine
    expose:
      - "80" # No host port mapping! Allows multiple instances.
    networks:
      - app-net
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost/"]
      interval: 10s
      timeout: 5s
      retries: 3

  # Public Facing Load Balancer
  loadbalancer:
    image: nginx:latest
    ports:
      - "8080:80"
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf:ro
    depends_on:
      - web
    networks:
      - app-net

networks:
  app-net:
    driver: bridge

---
