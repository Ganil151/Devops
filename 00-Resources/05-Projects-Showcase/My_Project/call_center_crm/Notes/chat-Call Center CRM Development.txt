### **3. JWT Token Verification**
JWT token verification should also be placed in the same utility file (`auth.js`) as the token generation. This ensures that both functions are grouped logically.

#### **Code Example: `backend/utils/auth.js`**

```javascript
const jwt = require('jsonwebtoken');

// Secret key for signing tokens (use environment variables in production)
const SECRET_KEY = 'your-secret-key';

  return jwt.sign({ id: user.id, email: user.email }, SECRET_KEY, { expiresIn: '1h' });

  verifyToken,
```

```javascript
const { verifyToken } = require('./utils/auth');

// Middleware to verify JWT token
const authenticateToken = (req, res, next) => {

  const token = authHeader && authHeader.split(' ')[1]; // Extract token from "Bearer <token>"

  const user = verifyToken(token);
```
